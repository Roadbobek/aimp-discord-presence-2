cmake_minimum_required( VERSION 3.28 )
project( aimp_discordPresence LANGUAGES CXX )

set( CMAKE_CXX_STANDARD 17 )
set( CMAKE_CXX_STANDARD_REQUIRED ON )

if( CMAKE_SIZEOF_VOID_P EQUAL 8 )
  set (X86 FALSE)
elseif( CMAKE_SIZEOF_VOID_P EQUAL 4 )
  set (X86 TRUE)
endif()

set( LIB_DIR ${CMAKE_CURRENT_SOURCE_DIR}/lib )
set( SRC_DIR ${CMAKE_CURRENT_SOURCE_DIR}/src )

add_compile_options( /W4 /WX )

add_subdirectory( ${SRC_DIR} )

# --- Unit tests ---------------------------------------------------------------

# Cover Payload Tests
# Offline tests for the pure Discord payload builders. The test binary compiles
# src/discord_ipc.cpp directly, so there is one implementation and no mock.
# No third-party framework and no network are involved.
enable_testing()

add_executable( payload_tests
  tests/payload_tests.cpp
  ${SRC_DIR}/discord_ipc.cpp
)
target_include_directories( payload_tests PRIVATE ${SRC_DIR} )
add_test( NAME payload_tests COMMAND payload_tests )

# Cover Publisher Tests
# Offline tests for the album art pipeline (fetch, resize, local fallback).
# The test binary compiles the real cover_publisher, album_art, and
# local_art sources directly — no mocks, no network, no third-party
# framework. winhttp is linked for the HTTP layer symbols but no
# requests are made at runtime.
add_executable( cover_publisher_tests
        tests/cover_publisher_tests.cpp
        ${SRC_DIR}/cover_publisher.cpp
        ${SRC_DIR}/album_art.cpp
        ${SRC_DIR}/local_art.cpp
)

target_include_directories( cover_publisher_tests PRIVATE
        ${LIB_DIR}/aimp-sdk/Sources/Cpp
        ${LIB_DIR}/aimp-glue
        ${SRC_DIR}
)

target_link_libraries( cover_publisher_tests PRIVATE
        winhttp
)

add_test( NAME cover_publisher_tests COMMAND cover_publisher_tests )

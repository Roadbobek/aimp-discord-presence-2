cmake_minimum_required( VERSION 3.28 )
project( aimp_discordPresence LANGUAGES CXX )

set( CMAKE_CXX_STANDARD 17 )
set( CMAKE_CXX_STANDARD_REQUIRED ON )

if( CMAKE_SIZEOF_VOID_P EQUAL 8 )
  set (X86 FALSE)
elseif( CMAKE_SIZEOF_VOID_P EQUAL 4 )
  set (X86 TRUE)
endif()

set( LIB_DIR ${CMAKE_CURRENT_SOURCE_DIR}/lib )
set( SRC_DIR ${CMAKE_CURRENT_SOURCE_DIR}/src )

add_compile_options( /W4 /WX )

add_subdirectory( ${SRC_DIR} )

# --- Unit tests -------------------------------------------------------------
#
# Offline tests for the pure Discord payload builders. The test binary compiles
# src/discord_ipc.cpp directly, so there is one implementation and no mock.
# No third-party framework and no network are involved.
enable_testing()

add_executable( payload_tests
  tests/payload_tests.cpp
  ${SRC_DIR}/discord_ipc.cpp
)
target_include_directories( payload_tests PRIVATE ${SRC_DIR} )
add_test( NAME payload_tests COMMAND payload_tests )
